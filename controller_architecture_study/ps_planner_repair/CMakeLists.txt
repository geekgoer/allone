cmake_minimum_required(VERSION 3.15)
project(ps_planner_profile LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
enable_testing()
if(NOT PAYLOAD OR NOT JSON_INCLUDE_DIR)
    message(FATAL_ERROR "PAYLOAD and JSON_INCLUDE_DIR are required")
endif()
foreach(variant baseline profile move_baseline move_profile)
    add_executable(replay_${variant} "${PAYLOAD}/${variant}/scheduler.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/profile_replay.cpp")
    target_include_directories(replay_${variant} PRIVATE "${PAYLOAD}/${variant}" "${CMAKE_CURRENT_SOURCE_DIR}" "${JSON_INCLUDE_DIR}")
    if(variant MATCHES "profile$")
        target_compile_definitions(replay_${variant} PRIVATE PS_PROFILE_ENABLED)
    endif()
    if(MSVC)
        target_compile_options(replay_${variant} PRIVATE /FI"${CMAKE_CURRENT_SOURCE_DIR}/windows_clock.hpp" /W0 /O2 /fp:strict)
    else()
        target_compile_options(replay_${variant} PRIVATE -O2 -ffp-contract=off)
        target_link_libraries(replay_${variant} PRIVATE pthread)
    endif()
endforeach()
add_test(NAME baseline_parity COMMAND replay_baseline "${PAYLOAD}/inputs.json" 1 "${CMAKE_CURRENT_BINARY_DIR}/baseline_test.json")
add_test(NAME instrumented_parity COMMAND replay_profile "${PAYLOAD}/inputs.json" 1 "${CMAKE_CURRENT_BINARY_DIR}/profile_test.json")
add_test(NAME move_parity COMMAND replay_move_baseline "${PAYLOAD}/inputs.json" 1 "${CMAKE_CURRENT_BINARY_DIR}/move_test.json")
add_test(NAME move_instrumented_parity COMMAND replay_move_profile "${PAYLOAD}/inputs.json" 1 "${CMAKE_CURRENT_BINARY_DIR}/move_profile_test.json")
