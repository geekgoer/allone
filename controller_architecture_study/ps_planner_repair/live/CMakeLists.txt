cmake_minimum_required(VERSION 3.15)
project(ps_management_repair LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
if(NOT JSON_INCLUDE_DIR)
    message(FATAL_ERROR "JSON_INCLUDE_DIR required")
endif()
add_library(management scheduler.cpp live_calendar.cpp revalidate.cpp)
target_include_directories(management PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${JSON_INCLUDE_DIR}")
if(MSVC)
    target_compile_options(management PRIVATE /FI"${CMAKE_CURRENT_SOURCE_DIR}/../windows_clock.hpp" /W0 /O2 /fp:strict)
else()
    target_compile_options(management PRIVATE -O2 -ffp-contract=off)
    target_link_libraries(management PUBLIC pthread)
endif()
enable_testing()
foreach(name scheduler_tests live_calendar_tests revalidate_tests)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE management)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W0)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endforeach()
add_executable(management_bench management_bench.cpp)
target_link_libraries(management_bench PRIVATE management)
add_executable(snapshot_precheck_bench snapshot_precheck_bench.cpp)
target_link_libraries(snapshot_precheck_bench PRIVATE management)
if(MSVC)
    target_compile_options(management_bench PRIVATE /FI"${CMAKE_CURRENT_SOURCE_DIR}/../windows_clock.hpp" /W0)
endif()
